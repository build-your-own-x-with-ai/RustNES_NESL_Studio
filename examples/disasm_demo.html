; NES ROM 反汇编
; PRG: 2 个 bank (32768 字节)
; CHR: 1 个 bank (8192 字节)
; Mapper: 0
; Reset 向量: $8000

L8000:  ; 代码块起始
    SEI
    CLD
    LDX #$40
    STX $4017
    LDX #$FF
    TXS
    INX
    STX $2000
    STX $2001
    STX $4010
    BIT $2002
    BIT $2002
    BPL L8017

L8017:  ; 初始化内存
    LDA #$00
    STA $00,X
    STA $0100,X
    STA $0300,X
    INX
    BNE L8017

L8056:  ; 主循环
    JSR $8CF6  ; wait_vblank
    JSR $8C76  ; joypad(0)
    STA $00     ; 保存手柄状态

    ; 检查左键
    LDA $00
    AND #$02    ; BTN_LEFT
    CMP #$00
    BEQ L806D
    DEC $02     ; x -= 1

L806D:
    ; 检查右键
    LDA $00
    AND #$01    ; BTN_RIGHT
    CMP #$00
    BEQ L807A
    INC $02     ; x += 1

L8B51:  ; 更新星星位置
    LDA $06     ; sy（星星 Y 坐标）
    CLC
    ADC $07     ; + speed
    STA $06
    CMP #$E0    ; 到达底部？
    BCC L8B64

    LDA $05     ; sx（星星 X 坐标）
    SEC
    SBC $02     ; - px（玩家 X 坐标）
    CMP #$10    ; 距离 < 16？
    BCS L8B7E
    LDA $06
    CMP #$C8    ; Y 坐标检查
    BCC L8B7E

    ; 碰撞发生！
    INC $04     ; score++
    JSR $8AB5  ; beep

命令行使用

# 反汇编 NES ROM
nesc disasm game.nes

# 输出到文件
nesc disasm game.nes -o game.asm

# 反汇编商业 ROM（本地使用）
nesc disasm SuperMarioBros.nes -o smb.asm
